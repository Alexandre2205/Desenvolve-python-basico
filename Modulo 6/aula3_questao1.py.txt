# Solicitar números indefinidamente
print("Digite números inteiros (pelo menos 4).")
print("Quando terminar, pressione Enter sem digitar nada.")

numeros = []

while True:
    entrada = input("Digite um número (ou Enter para parar): ")
    if entrada == "":
        break
    numeros.append(int(entrada))

# Verificar se o usuário digitou ao menos 4 números
if len(numeros) < 4:
    print("Você precisa digitar pelo menos 4 números.")
else:
    print("\nLista original:", numeros)
    print("3 primeiros elementos:", numeros[:3])       # do início até o índice 2
    print("2 últimos elementos:", numeros[-2:])        # últimos 2
    print("Lista invertida:", numeros[::-1])           # passo -1 inverte
    print("Elementos de índice par:", numeros[::2])    # índices 0, 2, 4...
    print("Elementos de índice ímpar:", numeros[1::2]) # índices 1, 3, 5...
