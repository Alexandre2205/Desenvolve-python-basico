# Passo 1: receber quantidade de elementos da lista 1
n1 = int(input("Digite a quantidade de elementos da lista 1: "))
print(f"Digite os {n1} elementos da lista 1:")
lista1 = [int(input()) for _ in range(n1)]

# Passo 2: receber quantidade de elementos da lista 2
n2 = int(input("Digite a quantidade de elementos da lista 2: "))
print(f"Digite os {n2} elementos da lista 2:")
lista2 = [int(input()) for _ in range(n2)]

# Passo 3: intercalar
lista_intercalada = []
min_len = min(n1, n2)

# intercala até acabar a menor lista
for i in range(min_len):
    lista_intercalada.append(lista1[i])
    lista_intercalada.append(lista2[i])

# adiciona o restante da maior lista
if n1 > n2:
    lista_intercalada.extend(lista1[min_len:])
else:
    lista_intercalada.extend(lista2[min_len:])

# Passo 4: mostrar resultado
print("Lista intercalada:", *lista_intercalada)
